cmake_minimum_required(VERSION 3.20)
project(CoD3Depth LANGUAGES CXX)
set(RESHADE_SDK "" CACHE PATH "Directory containing the ReShade 6.8.0 include headers")
if(NOT EXISTS "${RESHADE_SDK}/reshade.hpp")
    message(FATAL_ERROR "Set RESHADE_SDK to the ReShade 6.8.0 include directory")
endif()
add_library(CoD3Depth SHARED depth.cpp)
target_compile_features(CoD3Depth PRIVATE cxx_std_20)
target_compile_definitions(CoD3Depth PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(CoD3Depth PRIVATE "${RESHADE_SDK}")
target_link_libraries(CoD3Depth PRIVATE d3d11 d3dcompiler)
set_target_properties(CoD3Depth PROPERTIES PREFIX "" SUFFIX ".addon64")
